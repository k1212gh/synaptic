-- 단일 SQL로 전체 유저 엣지 계산 CREATE FUNCTION compute_edges_batch(uid uuid, k int, threshold float) RETURNS TABLE(from_id uuid, to_id uuid, sim float) AS $$ SELECT c1.id, c2.id, 1-(c1.embedding <=> c2.embedding) FROM chunks c1, chunks c2 JOIN pages p ON p.id = c1.page_id WHERE p.user_id = uid AND c1.id < c2.id AND 1-(c1.embedding <=> c2.embedding) > threshold ORDER BY sim DESC LIMIT k * (SELECT count(*) FROM chunks...) $$ LANGUAGE sql;
